\newacronym{vrp}{VRP}{Vehicle Routing Problem}
\newacronym{mpvrpcc}{MPVRP-CC}{Multi-Product Vehicle Routing Problem with Changeover Costs}
\newacronym{milp}{MILP}{Mixed-Integer Linear Programming}
\newacronym{cp}{CP}{Constraint Programming}
\newacronym{tsp}{TSP}{Traveling Salesman Problem}

\newglossaryentry{changeover-cost}{
  name={changeover cost},
  description={Cost incurred when a vehicle changes from carrying one product
  to carrying another product}
}

\newglossaryentry{delivery-piece}{
  name={delivery piece},
  description={Portion of a request created when its demand is split}
}

\newglossaryentry{mini-trip}{
  name={mini-trip},
  description={Unit constructed by the hybrid method, consisting of one loading
  operation followed by deliveries of the loaded product}
}

\newglossaryentry{request}{
  name={request},
  description={Station--product pair with positive demand}
}

\newglossaryentry{sequence-variable}{
  name={sequence variable},
  description={Constraint programming variable representing membership and
  ordering within a sequence}
}

\newglossaryentry{supply-node}{
  name={supply node},
  description={Depot--product pair with available stock}
}
